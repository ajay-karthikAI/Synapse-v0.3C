# Synapse — A Zenith Company
# Requirements for the LEGACY Streamlit application (app.py + Data/Retrieval/Generation).
# Install with: pip install -r requirements.txt
#
# The `synapse` package itself does NOT depend on any of this — see pyproject.toml,
# whose only runtime dependency is pydantic. That separation is deliberate: the
# schema, corpus, governance and evaluation layers must be importable and testable
# without a vector-search or UI stack installed.

# Core AI / LLM
openai==2.32.0                    # Chat completions (answer generation, reranking) AND embeddings

# Embeddings + Vector Search
faiss-cpu==1.13.2                 # Local similarity search over the embedding matrix
numpy==2.4.3                      # FAISS's array interface

# Keyword Search
rank-bm25==0.2.2                  # BM25 — the only retrieval path that works without an API key

# Chunking
langchain-text-splitters==1.1.2   # RecursiveCharacterTextSplitter only. Pulls langchain-core transitively; we do not pin it because we do not import it.

# UI
streamlit==1.56.0                 # The patient-facing app

# Environment
python-dotenv==1.2.2              # Loads OPENAI_API_KEY / NCBI_API_KEY from .env
